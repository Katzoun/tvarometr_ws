# Python deps for the inference container.

# check torch.cuda.get_arch_list() before bumping. +cu121 picks the CUDA wheel.
--extra-index-url https://download.pytorch.org/whl/cu121

torch==2.5.1+cu121
torchvision==0.20.1+cu121

# ROS Humble's apt Python modules are built against numpy 1 (_ARRAY_API errors on 2).
numpy<2
opencv-python-headless
pillow
scipy
pandas
tqdm
ultralytics

# The yolov8x checkpoint needs omegaconf, but ultralytics doesn't declare it and
# just pip-installs it on first load. We turn that off, so it has to be here.
omegaconf

# The vendored MiVOLO code imports timm internals that later versions renamed -
# remap_checkpoint is gone from 0.9.x on. Same pin MiVOLO upstream uses.
timm==0.8.13.dev0
